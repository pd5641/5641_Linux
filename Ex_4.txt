# przejdź do katalogu domowego
cd ~

# przejdź do katalogu Moje Katalogi
cd "Moje Katalogi"

# utwórz plik plik.txt z podaną zawartością
echo "To jest zawartość pliku" > plik.txt

# utwórz dowiązanie twarde
ln plik.txt plik_twardy.txt

# sprawdź dowiązanie twarde
ls -li

# przenieś oryginalny plik do katalogu domowego
mv plik.txt ~

# sprawdź zawartość dowiązania twardego
cat plik_twardy.txt

# wróć do katalogu domowego
cd ~

# utwórz dowiązanie symboliczne
ln -s ~/plik.txt link_symboliczny.txt

# sprawdź dowiązanie symboliczne
ls -l link_symboliczny.txt

# usuń oryginalny plik
rm plik.txt

# sprawdź dowiązanie symboliczne po usunięciu pliku
cat link_symboliczny.txt

# sprawdź, czy dowiązanie twarde nadal działa
cat "Moje Katalogi/plik_twardy.txt"
