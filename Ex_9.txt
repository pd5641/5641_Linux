cd ~

# utwórz plik z przykładowymi danymi
echo -e "Ala\nKot\nPies\nAla\nPies\nZebra\nZebra\n" > przykladowe_dane2.txt

# posortuj dane alfabetycznie i zapisz do nowego pliku
sort przykladowe_dane2.txt > posortowane_dane.txt

# wyświetl posortowane dane
cat posortowane_dane.txt

# output do weryfikacji

Ala
Ala
Kot
Pies
Pies
Zebra
Zebra

# usuń duplikaty z posortowanego pliku
uniq posortowane_dane.txt > unikalne_dane.txt

# wyświetl dane bez duplikatów
cat unikalne_dane.txt

# output do weryfikacji

Ala
Kot
Pies
Zebra
